# CUGA ARC-3 MVP-1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an ARC-AGI-3 runner that compares scaffold stages (S0-S3), reports win-rate lift on a fixed 50-game subset, and exposes planning/reasoning/action/reflection in a right-hand UI panel.

**Architecture:** Use a Python monorepo with `uv`, `pytest`, and `streamlit`. Keep ARC API interaction in a dedicated adapter, run scaffold logic through a deterministic loop, persist run events as JSONL, and drive both evaluation and UI from the same event schema for attribution and replay consistency.

**Tech Stack:** Python 3.12, `uv`, `pytest`, `httpx`, `pydantic`, `streamlit`, `python-dotenv`, JSONL artifacts.

```python
# src/cuga_arc3/arc_adapter/client.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Any

import httpx


@dataclass
class ArcClient:
    base_url: str
    api_key: str
    transport: httpx.AsyncBaseTransport | None = None

    def _headers(self) -> dict[str, str]:
        return {"Authorization": f"Bearer {self.api_key}"}

    async def list_games(self) -> list[dict[str, Any]]:
        async with httpx.AsyncClient(base_url=self.base_url, transport=self.transport, timeout=30.0) as client:
            response = await client.get("/games", headers=self._headers())
            response.raise_for_status()
            return response.json()
```

```python
# src/cuga_arc3/runner/loop.py
from __future__ import annotations

from dataclasses import dataclass

from cuga_arc3.runner.events import TraceEvent
from cuga_arc3.runner.scaffolds import StageProfile


@dataclass
class RunnerLoop:
    stage_profile: StageProfile
    max_steps: int

    def simulate(self, game_id: str) -> list[TraceEvent]:
        events: list[TraceEvent] = []
        events.append(
            TraceEvent(
                event_type="plan_updated",
                stage=self.stage_profile.stage,
                game_id=game_id,
                step_idx=0,
                payload={"plan": "initial plan"},
            )
        )
        for idx in range(1, self.max_steps + 1):
            events.append(
                TraceEvent(
                    event_type="action_proposed",
                    stage=self.stage_profile.stage,
                    game_id=game_id,
                    step_idx=idx,
                    payload={"action": 1, "rationale": "default action policy"},
                )
            )
            if self.stage_profile.reflection_enabled:
                events.append(
                    TraceEvent(
                        event_type="reflection",
                        stage=self.stage_profile.stage,
                        game_id=game_id,
                        step_idx=idx,
                        payload={"note": "adjust next action from observation"},
                    )
                )
        events.append(
            TraceEvent(
                event_type="episode_end",
                stage=self.stage_profile.stage,
                game_id=game_id,
                step_idx=self.max_steps,
                payload={"status": "terminated"},
            )
        )
        return events
```

```python
# src/cuga_arc3/eval/manifest.py
from dataclasses import dataclass


@dataclass(frozen=True)
class BenchmarkManifest:
    core: list[str]
    stress: list[str]

    @property
    def total_games(self) -> int:
        return len(self.core) + len(self.stress)

    @classmethod
    def from_dict(cls, payload: dict) -> "BenchmarkManifest":
        core = payload.get("core", [])
        stress = payload.get("stress", [])
        manifest = cls(core=core, stress=stress)
        if len(core) != 30 or len(stress) != 20:
            raise ValueError("Manifest must contain 30 core and 20 stress games")
        return manifest
```

```python
from cuga_arc3.eval.manifest import BenchmarkManifest


@dataclass(frozen=True)
class StageMetrics:
    stage: str
    solved_count: int
    total_games: int

    @property
    def win_rate(self) -> float:
        return self.solved_count / self.total_games if self.total_games else 0.0


def compute_stage_metrics(stage: str, solved_game_ids: list[str], manifest: BenchmarkManifest) -> StageMetrics:
    solved_count = len(set(solved_game_ids))
    return StageMetrics(stage=stage, solved_count=solved_count, total_games=manifest.total_games)
```

```python
# src/cuga_arc3/ui/app.py
from __future__ import annotations

from typing import Any

import streamlit as st

from cuga_arc3.runner.events import TraceEvent


def project_right_panel(events: list[TraceEvent]) -> dict[str, Any]:
    plan = ""
    action: dict[str, Any] = {}
    reflection = ""
    for event in events:
        if event.event_type == "plan_updated":
            plan = str(event.payload.get("plan", ""))
        elif event.event_type == "action_proposed":
            action = event.payload
        elif event.event_type == "reflection":
            reflection = str(event.payload.get("note", ""))
    return {"plan": plan, "action": action, "reflection": reflection}


def render(events: list[TraceEvent]) -> None:
    left, right = st.columns([2, 1])
    with left:
        st.subheader("Run Overview")
        st.write("ARC-3 stage execution")
    with right:
        st.subheader("Planning + Reasoning")
        panel = project_right_panel(events)
        st.markdown("**Plan**")
        st.write(panel["plan"])
        st.markdown("**Action**")
        st.json(panel["action"])
        st.markdown("**Reflection**")
        st.write(panel["reflection"])
```

```python
# src/cuga_arc3/reporting.py
def stage_lift(baseline: dict, candidate: dict) -> dict:
    absolute_lift = round(candidate["win_rate"] - baseline["win_rate"], 2)
    if baseline["win_rate"] == 0:
        relative = 0.0
    else:
        relative = round((absolute_lift / baseline["win_rate"]) * 100, 2)
    return {
        "from_stage": baseline["stage"],
        "to_stage": candidate["stage"],
        "absolute_lift": absolute_lift,
        "relative_lift_pct": relative,
    }
```
